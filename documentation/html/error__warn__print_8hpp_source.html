72/**
73 * @brief throw_error is a custom error logger to the terminal. This will WILL throw.
74 * @param error Your type of error.
75 * @param error_option An optional string to be added at the start, could be a path, text...
76 * @throw ErrorException
77 */
78inline void throw_error(ErrorType const error, const std::string& error_option = "")
79{
80 switch (error)
81 {
82 case ErrorType::NO_ERROR:
83 break;
84 case ErrorType::VALUE_ERROR:
85 throw ErrorException(ErrorType::VALUE_ERROR, error_option + " <- " + ERR_STRING::VALUE);
86 break;
87 case ErrorType::SYNTAX_ERROR:

89 break;
90 case ErrorType::OPTION_UNAVAILABLE:
91 throw ErrorException(ErrorType::OPTION_UNAVAILABLE, error_option + " <- " + ERR_STRING::OPTION_UNAVAILABLE);
92 break;
93 case ErrorType::PATH_NOT_FOUND:
94 throw ErrorException(ErrorType::PATH_NOT_FOUND, error_option + " <- " + ERR_STRING::PATH_NOT_FOUND);
95 break;
96 case ErrorType::PATH_NOT_ACCESSIBLE:
97 throw ErrorException(ErrorType::PATH_NOT_ACCESSIBLE, error_option + " <- " + ERR_STRING::PATH_NOT_ACCESSIBLE);
98 break;
99 case ErrorType::PATH_INVALID:
100 throw ErrorException(ErrorType::PATH_INVALID, error_option + " <- " + ERR_STRING::PATH_INVALID);
101 break;
102 case ErrorType::FILE_INVALID:
103 throw ErrorException(ErrorType::FILE_INVALID, error_option + " <- " + ERR_STRING::FILE_INVALID);
104 break;
105 case ErrorType::FILE_CORRUPTED:
106 throw ErrorException(ErrorType::FILE_CORRUPTED, error_option + " <- " + ERR_STRING::FILE_CORRUPTED);
107 break;
108 case ErrorType::DRIVE_ERROR:
109 throw ErrorException(ErrorType::DRIVE_ERROR, error_option + " <- " + ERR_STRING::DRIVE_ERROR);
110 break;
111 case ErrorType::INVALID_DECOMPRESSION:
112 throw ErrorException(ErrorType::INVALID_DECOMPRESSION, error_option + " <- " + ERR_STRING::INVALID_DECOMPRESSION);
113 break;
114 case ErrorType::MISSING_ARGUMENT:
115 throw ErrorException(ErrorType::MISSING_ARGUMENT, error_option + " <- " + ERR_STRING::MISSING_ARGUMENT);
116 break;
117 case ErrorType::DIR_COMPRESSION:
118 throw ErrorException(ErrorType::DIR_COMPRESSION, error_option + " <- " + ERR_STRING::DIR_COMPRESSION);
119 break;
120 default:
121 assert(false);
122 break;
123 }
