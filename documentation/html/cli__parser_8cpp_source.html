39/**
40 * @brief Semantically check an argument (e.g. -preset 8)
41 * @param option The argument
42 * @return The token
43 */
44std::expected<Token, ErrorType> lex(const std::string& option)
45{
46 auto token_string = option.substr(0, option.find(' '));
47
48 std::string token_value{};
49 if (auto value_pos = option.find_first_not_of(' ', option.find(' ')); value_pos != std::string::npos)//handles flags
50 token_value = option.substr(value_pos, std::string::npos);
51
52
53 TokenType token_type;
54
55 //Find TokenType by checking if token_string is equal to any command
56 if (token_string == token_strings[static_cast<size_t>(TokenType::COMPRESSION_PRESET)]) token_type = TokenType::COMPRESSION_PRESET;
57 else if (token_string == token_strings[static_cast<size_t>(TokenType::FILENAME_IN)]) token_type = TokenType::FILENAME_IN;
58 else if (token_string == token_strings[static_cast<size_t>(TokenType::FILENAME_OUT)]) token_type = TokenType::FILENAME_OUT;
59 else if (token_string == token_strings[static_cast<size_t>(TokenType::N_FILES)]) token_type = TokenType::N_FILES;
60 else if (token_string == token_strings[static_cast<size_t>(TokenType::SIZE_FILES)]) token_type = TokenType::SIZE_FILES;
61 else if (token_string == token_strings[static_cast<size_t>(TokenType::LONG_HELP)]) token_type = TokenType::LONG_HELP;
62 else if (token_string == token_strings[static_cast<size_t>(TokenType::HELP)]) token_type = TokenType::HELP;
63 else if (token_string == token_strings[static_cast<size_t>(TokenType::FORCE_COMPRESSION)]) token_type = TokenType::FORCE_COMPRESSION;
64 else if (token_string == token_strings[static_cast<size_t>(TokenType::DELETE_INPUT)]) token_type = TokenType::DELETE_INPUT;
65 else if (token_string == token_strings[static_cast<size_t>(TokenType::COMPRESSOR_TYPE)]) token_type = TokenType::COMPRESSOR_TYPE;
66 else if (token_string == token_strings[static_cast<size_t>(TokenType::CONCATENATE)]) token_type = TokenType::CONCATENATE;
67 else if (token_string == token_strings[static_cast<size_t>(TokenType::DO_DECOMP_AFTER_CONCAT)]) token_type = TokenType::DO_DECOMP_AFTER_CONCAT;
68 else if (token_string == token_strings[static_cast<size_t>(TokenType::DO_NOT_DECOMP_AFTER_CONCAT)]) token_type = TokenType::DO_NOT_DECOMP_AFTER_CONCAT;
69 else if (token_string == token_strings[static_cast<size_t>(TokenType::CONCURRENCY)]) token_type = TokenType::CONCURRENCY;
70 else
71 throw_error(ErrorType::SYNTAX_ERROR, token_string);
72
73 //handle flags
74 switch (token_type)
75 {
76 case TokenType::DELETE_INPUT:
77 case TokenType::FORCE_COMPRESSION:
78 case TokenType::CONCATENATE:
79 case TokenType::HELP:
80 case TokenType::LONG_HELP:
81 return Token{ token_type, true };
82 break;
83 default:
84 break;
85 }
86
87
88 ValueType value;
89 //Find token_value and do error checking
90 switch (token_type)
91 {
92 case TokenType::FILENAME_IN:
93 {
94 //Test if we can find and access the input file
95 if (!fs::exists(token_value))
96 throw_error(ErrorType::PATH_NOT_FOUND, token_value);
97
98 if (std::ifstream file(token_value); !file.is_open())
99 throw_error(ErrorType::PATH_NOT_ACCESSIBLE, token_value);
100
101 value.emplace<fs::path>(token_value);
102 }
103 break;
104 case TokenType::FILENAME_OUT:
105 {
106 fs::path path{ token_value };
107
108 //Test if we can write a file to the given dir
109 auto test_path = path.parent_path() / "test_writeability";
110 std::ofstream test_file{test_path};
111 if (!test_file.is_open())
112 {
113 fs::remove(test_path);
114 throw_error(ErrorType::PATH_NOT_ACCESSIBLE, path.string());
115 }
116 test_file.close();
117 fs::remove(test_path);
118
119 path.replace_extension(FILE_EXTENSION);
120
121 value.emplace<fs::path>(path);
122 }
123 break;
124 case TokenType::COMPRESSOR_TYPE:
125 {
126 std::transform(token_value.begin(), token_value.end(), token_value.begin(), [](char c) { return std::toupper(c); });
127
128 for (int i = 0; i < static_cast<int>(CompType::MAX); i++)
129 {
130 if (COMPRESSOR_STR_OPTIONS[i] == token_value)
131 {
132 value.emplace<size_t>(static_cast<size_t>(i));
133 break;
134 }
135 }
136
137 if (std::holds_alternative<std::monostate>(value))
138 throw_error(ErrorType::SYNTAX_ERROR, token_string + token_value);
139 }
140 break;
141 case TokenType::COMPRESSION_PRESET:
142 {
143 //Convert and check the compression preset
144 size_t value_cmpr;
145 auto res = std::from_chars(token_value.data(), token_value.data() + token_value.size(), value_cmpr);
146
147 if (res.ec != std::errc() or (value_cmpr > CompPreset::COMP_MAX or value_cmpr < CompPreset::NO_COMP))//if not a number or outside of our range
148 throw_error(ErrorType::VALUE_ERROR, token_string + token_value);
149
150 value.emplace<size_t>(value_cmpr);
151 }
152 break;
153 case TokenType::N_FILES:
154 {
155 int value_temp;
156 auto res = std::from_chars(token_value.data(), token_value.data() + token_value.size(), value_temp);
157
158 if (res.ec != std::errc() or (value_temp > N_FILES_LIMIT or value_temp < 1))//if less than 1 or over N_FILES_LIMIT
159 throw_error(ErrorType::VALUE_ERROR, token_string + token_value);
160
161 value.emplace<size_t>(value_temp);
162 }
163 break;
164 case TokenType::SIZE_FILES:
165 {
166 int value_temp;
167 auto res = std::from_chars(token_value.data(), token_value.data() + token_value.size(), value_temp);
168
169 if (res.ec != std::errc() or (value_temp < SIZE_FILES_MIN))//if negative or under SIZE_FILES_MIN
170 throw_error(ErrorType::VALUE_ERROR, token_string + token_value);
171
172 value.emplace<size_t>(value_temp);
173 }
174 break;
175 case TokenType::CONCURRENCY:
176 {
177 size_t value_temp;
178 auto const res = std::from_chars(token_value.data(), token_value.data() + token_value.size(), value_temp);
179
180 if (res.ec != std::errc())
181 throw_error(ErrorType::VALUE_ERROR, token_string + token_value);
182
183 if (value_temp < 0)
184 print_warn(WarningType::CONCURRENCY_OUT_OF_RANGE_LOWER, token_string + token_value);
185
186 if (value_temp > std::thread::hardware_concurrency())
187 print_warn(WarningType::CONCURRENCY_OUT_OF_RANGE_UPPER, token_string + token_value);
188
189 //I had no fucking idea size_t had suffix 'uz'
190 value_temp = std::clamp(value_temp, 0uz, static_cast<size_t>(std::thread::hardware_concurrency()));
191
192 value.emplace<size_t>(value_temp);
193 }
194 break;
195 //the next are all flags handled by the switch just above
196 case TokenType::LONG_HELP:
197 case TokenType::HELP:
198 case TokenType::FORCE_COMPRESSION:
199 case TokenType::DELETE_INPUT:
200 case TokenType::CONCATENATE:
201 case TokenType::DO_DECOMP_AFTER_CONCAT:
202 case TokenType::DO_NOT_DECOMP_AFTER_CONCAT:
203 case TokenType::NO_TYPE:
204 break;
205 }
206
207 return Token{ token_type, value };
208}

210/**
211 * @brief Read the cli.
212 * @param argc Number of cli arguments, where an argument is a 'word' separated by spaces, e.g. -i or -preset
213 * @param argv Pointer to the arguments
214 * @return The options obtained from the cli
215 *
216 * @throws ErrorException for any kind of syntactic or semantic error
217 * @throws HelpException if help command was used
218 * @todo simplify whatever spaghetti is inside of this
219 */
220Options parse(int argc, char* argv[])
221{
222 if (argc == 1)
223 throw_error(ErrorType::SYNTAX_ERROR, "No arguments given.");
224
225 Options options;
226 //Parse the cli
227 //starting at 1 means we ignore the first (name of program)
228 for (int index = 1; index < argc; index++)
229 {
230 std::string option{ argv[index] };
231 option += ' ';
232 if (index + 1 < argc)//stay in bounds
233 option += argv[++index];
234
235 //If the path has spaces, this makes sure that we take the whole path (we check for ")
236 //else we throw
237 if (std::ranges::count(option, '\"') == 1)
238 {
239 bool valid = false;
240
241 for(;index < argc - 1; index++)//keep parsing arguments until we find an ending double quote
242 {
243 std::string option_temp{ argv[index + 1] };
244 option += option_temp;
245 if (std::count(option_temp.begin(), option_temp.end(), '\"') == 1)
246 {
247 valid = true;
248 break;
249 }
250 }
251
252 if (!valid)
253 throw_error(ErrorType::PATH_INVALID, option);
254 }
255
256 auto token = lex(option);
257
258 if (token)
259 {
260 switch (token.value().get_type())
261 {
262 case TokenType::FILENAME_IN:
263 options.filename_in = std::get<fs::path>(token.value().get_value());
264 break;
265 case TokenType::FILENAME_OUT:
266 options.filename_out = std::get<fs::path>(token.value().get_value());
267 break;
268 case TokenType::COMPRESSOR_TYPE:
269 options.preset = std::get<size_t>(token.value().get_value());
270 break;
271 case TokenType::COMPRESSION_PRESET:
272 options.preset = std::get<size_t>(token.value().get_value());
273 break;
274 case TokenType::N_FILES:
275 options.n_files = std::get<size_t>(token.value().get_value());
276 break;
277 case TokenType::SIZE_FILES:
278 options.size_files = std::get<size_t>(token.value().get_value());
279 break;
280 case TokenType::CONCURRENCY:
281 options.concurrency = std::get<size_t>(token.value().get_value());
282 break;
283 //DO index-- FOR FLAGS, it stops us from skipping arguments
284 case TokenType::HELP:
285 case TokenType::LONG_HELP:
286 options.need_help = std::get<bool>(token.value().get_value());
287 if (index + 1 < argc)
288 index--;
289 break;
290 case TokenType::FORCE_COMPRESSION:
291 options.force_compression = std::get<bool>(token.value().get_value());
292 if (index + 1 < argc)
293 index--;
294 break;
295 case TokenType::DELETE_INPUT:
296 options.delete_input = std::get<bool>(token.value().get_value());
297 if (index + 1 < argc)
298 index--;
299 break;
300 case TokenType::CONCATENATE:
301 options.concatenate_files = std::get<bool>(token.value().get_value());
302 if (index + 1 < argc)
303 index--;
304 break;
305 case TokenType::DO_DECOMP_AFTER_CONCAT:
306 options.decomp_after_concat = std::get<bool>(token.value().get_value());
307 if (index + 1 < argc)
308 index--;
309 break;
310 case TokenType::DO_NOT_DECOMP_AFTER_CONCAT:
311 if (index + 1 < argc)
312 index--;
313 break;
314 default:
315 std::terminate();
316 break;
317 }
318
319 }
320 else
321 {
322 throw_error(token.error(), option);
323 }
324 }
325
326 if (options.need_help)
327 {
328 std::println("{}", help_str);
329 return options;
330 }
331
332
333 if (fs::is_directory(options.filename_in) and !options.concatenate_files)//tried de/compressing a dir
334 throw_error(ErrorType::DIR_COMPRESSION, options.filename_in.string());
335 else if (options.filename_in.has_filename() and options.concatenate_files)//tried concatenating a file
336 throw_error(ErrorType::PATH_INVALID, options.filename_in.string());
337
338 //swap ifs
339 if (options.filename_in.empty() or options.filename_in.string().find_first_not_of(' ') == std::string::npos)//if filename_in is empty
340 {
341 if (options.concatenate_files)
342 options.filename_in = fs::current_path();
343 else if (!options.need_help)
344 throw_error(ErrorType::MISSING_ARGUMENT, token_strings[static_cast<int>(TokenType::FILENAME_IN)].data());
345 else
346 throw HelpException();
347 }
348
349 if (options.filename_out == DEFAULT_OUT_PATH)
350 {
351 options.filename_out = options.filename_in.parent_path();
352 options.filename_out /= DEFAULT_OUT_PATH;
353 options.filename_out.replace_extension(FILE_EXTENSION);
354 }
355
356 if (options.filename_in == options.filename_out)
357 {
358 options.filename_out.replace_filename(options.filename_out.stem().string() + "_out" + FILE_EXTENSION);
359 }
360
361 return options;
362}
