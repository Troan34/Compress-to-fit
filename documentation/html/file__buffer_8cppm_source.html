18/**
19 * @brief Ring queue-like where if size > capacity, the oldest elements are written to stream.
20 * At least the most recent #minimum_size_ number of elements shall be available for reading.
21 * @tparam T type stored, must be trivially copyable
22 */
23export template <typename T>
24 requires SerializableToDisk<std::ranges::range_value_t<T>> or SerializableToDisk<T>
25class ConcFileBuffer
26{
27public:
28
29 ConcFileBuffer(File& out_stream, size_t const minimum_size)
30 :out_stream_(out_stream), minimum_size_(minimum_size)
31 {
32 reserve(1.5f * minimum_size_);
33 }
34
35 ~ConcFileBuffer()
36 {
37 auto const elements_to_write_n = index_ - written_index_;
38 out_stream_.write(reinterpret_cast<char const*>(&data_[written_index_]), elements_to_write_n * sizeof(T));
39 allocator.deallocate(data_, capacity_);
40 }
41
42 /**
43 * @brief
44 * @param value
45 * @pre #index_ > #minimum_size_
46 */
47 constexpr void push(T const& value)
48 {
49 std::unique_lock lock{mut};
50
51 data_[true_index()] = value;
52 index_++;
53
54 write_buffer();
55 }
56
57 /**
58 * @brief
59 * @param value
60 * @pre #index_ > #minimum_size_
61 */
62 constexpr void push(T&& value)
63 {
64 std::unique_lock lock{mut};
65
66 data_[true_index()] = value;
67 index_++;

69 write_buffer();

72 /**
73 * @brief Appends a range to the container
74 * @param rg range to append
75 * @pre #index_ > #minimum_size_
76 */
77 template <std::ranges::range R>
78 requires std::convertible_to<std::ranges::range_reference_t<R>, T>
79 constexpr void append_range(R&& rg)
80 {
81 auto size = rg.end() - rg.start();

84

97 /**
98 * @brief Increase capacity to at least #new_capacity
99 * @param new_capacity New capacity of the buffer
100 */
101 constexpr void reserve(size_t new_capacity) noexcept(false)
102 {
103 if (new_capacity <= capacity_) return;
104
105 //allocate size for new data
106 T* new_data_ = allocator.allocate(new_capacity);
107 auto const size = index_ - written_index_;
108 std::unique_lock<std::shared_mutex> lock(mut);
109
110 if constexpr (std::is_trivially_copyable_v<T>)//if trivially copyable use std::memcpy
111 {
112 std::memcpy(new_data_, data_, size * sizeof(T));
113 }
114
115 /* To be used if we ever want to use non-trivial types
116 else//if not, call T's ctor
117 {
118 for (size_t i{}; i < size_; i++)
119 {
120 new (&new_data_[i]) T(std::move(data_[i]));
121
122 data_[i].~T();
123 }
124 }*/
125
126 data_ = new_data_;
127 auto old_cap = capacity_;
128 lock.unlock();
129
130 allocator.deallocate(data_, old_cap);
131 }

134 {
135 //if in debug mode

138 //holy yap

156 void set_minimum_size(size_t const new_minimum_size) noexcept

163private:
164 File& out_stream_;
165 T* data_;
166 std::allocator<T> allocator;
167 mutable std::shared_mutex mut;
168 size_t index_ = 0;
169 size_t written_index_ = 0; //oldest index that has not yet been written to file
170 size_t capacity_ = 0;
171 size_t minimum_size_ = 0;
172
173 /**
174 * @brief Writes elements outside the minimum range
175 * @pre lock MUST be locked with #lock
176 */
177 auto constexpr write_buffer() noexcept(false)
178 {
179 if (index_ + 1 - written_index_ > capacity_)//if we need to write elements to disk
180 {
181 auto const elements_to_write_n = index_ - minimum_size_ - written_index_;
182
183 //TODO: this blocks a thread, but if there are 12 threads AND they aren't writing frequently, it shouldn't be THAT bad. Benchmark this.
184 //We could try coroutines here
185 //Maybe we can lock the stream, or stuff like that
186 if constexpr (std::is_trivially_copyable_v<T>)
187 {
188 out_stream_.write(std::span{&data_[written_index_], elements_to_write_n * sizeof(T)});
189 }
190 else//we expect write_to to have been defined as per SerializableToDisk
191 {
192 std::ranges::for_each(std::views::counted(data_, elements_to_write_n),
193 [this](T const& value)
194 {
195 value.write_to(out_stream_.get_ref_out_stream());
196 }
197 );
198 }
199
200 written_index_ = index_ - minimum_size_;
201 }
202 }

205 [[nodiscard]] constexpr size_t true_index() const noexcept { return index_ % capacity_; }
206};
