162/**
163 * @brief Show a progress bar in terminal.
164 * @param options Receive miscellaneus data.
165 * @param progress From 0 to 1
166 */
167export void show_progress(const parser::Options& options = {}, float progress = 0.f, bool compressing = false)
168{
169 progress = std::min(1.F, progress);
170 constexpr int max_bar_width = 50;
171
172 std::print("\r\033[K");
173 std::print("\033[34mProgress on the ");
174 if (compressing)
175 std::print("compression:\033[0m");
176 else
177 std::print("extraction:\033[0m");
178
179
180 std::print("[\033[32m");
181 int const filled_width = std::ceil(max_bar_width * progress);
182 for (int i = 0; i < filled_width; i++)
183 {
184 std::print("█");
185 }
186 for (int i = filled_width; i < max_bar_width; i++)
187 {
188 std::print("-");
189 }
190
191
192 std::print("\033[0m] {:.1f}%", progress * 100.F);
193 std::cout << std::flush;
194}
