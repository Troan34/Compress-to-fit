181/**
182 * @brief Counts the number of consecutive equal elements from the start of two spans, up to a specified maximum.
183 * @tparam Type The element type in the spans, Type must support equality comparison.
184 * @param str1 The first sequence to compare.
185 * @param str2 The second sequence to compare.
186 * @param max_match Maximum number of elements to compare. Comparison stops when this limit is reached.
187 * @return The count of initial consecutive elements that are equal in both spans, constrained by the shorter span length and max_match.
188 */
189export template <typename Type>
190[[nodiscard]] constexpr auto count_equal(std::span<Type> const str1, std::span<Type> const str2, size_t const max_match)
191{
192 auto cond = std::min(str1.size(), str2.size());
193 size_t i = 0;
194 for (; i < cond and i < max_match; i++)
195 {
196 if (str1[i] != str2[i])
197 break;
198 }
199 return i;
200}

216concept SerializableToDisk = std::is_trivially_copyable_v<T> or MemberWriteable<T> or FreeWriteable<T>;

218template<SerializableToDisk T>
219void do_write_to(T const& t, std::ofstream& file)
220{
221 if constexpr (std::is_trivially_copyable_v<T> and !MemberWriteable<T> and !FreeWriteable<T>)
222 {
223 file.write(reinterpret_cast<char const *>(&t), sizeof(T));
224 }
225 else if constexpr (MemberWriteable<T>)
226 {
227 t.write_to(file);
228 }
229 else
230 {
231 write_to(t, file);
232 }
233}
