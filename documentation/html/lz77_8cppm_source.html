140 * @brief Slide the window

214 /**
215 * @pre @p data.max_size() > #MAX_HASH_SIZE
216 */
217 Rabin(const Window& data, uint32_t pos, CompPreset preset)
218 :data(data), position(pos), MAX_CHAIN(get_max_chain_from_preset(preset))
219 {
220 //compiler will probably use a memset-like optimization
221 for (int i = 0; i < MAX_HASH_SIZE; i++)
222 poss_table[i] = -1;
223
224 for (int i = 0; i < MAX_WINDOW_SIZE; i++)
225 prev_poss_table[i] = -1;
226
227 hash = data[0];
228 for (int i = 1; i < MIN_MATCH and i < data.get_data().size(); i++)//initialize the hash
229 {
230 hash = (static_cast<uint64_t>(hash) * BASE + data[position + i]) % MOD;
231 }
232 poss_table[bucket_index(hash)] = position;

234 }

336 static auto extract_header(std::span<std::byte const> const data)
337 {
338 constexpr size_t len1 = sizeof(decltype(compressed_length_));
339 constexpr size_t len2 = sizeof(decltype(uncompressed_length_));
340
341 if (data.size() < len1 + len2)
342 throw std::out_of_range("LZ77 block size is too small");
343
344 decltype(compressed_length_) compressed_length;
345 decltype(uncompressed_length_) uncompressed_length;
346
347 std::memcpy(&compressed_length, data.data(), len1);//get compressed_length from data
348 std::memcpy(&uncompressed_length, data.data() + len1, len2);//get uncompressed_length from data
349
350 return std::make_pair(compressed_length, uncompressed_length);
351 }
352
353
354 explicit LZ77Block(std::span<std::byte const> const data, fs::path const& file, Alloc const& alloc = Alloc())
355 : buffer_(alloc)
356 {
357 constexpr size_t len1 = sizeof(decltype(compressed_length_));
358 constexpr size_t len2 = sizeof(decltype(uncompressed_length_));
359
360 auto const [compressed_length__, uncompressed_length__] = extract_header(data);
361 compressed_length_ = compressed_length__;
362 uncompressed_length_ = uncompressed_length__;
363
364 if (len1 + len2 + compressed_length_ > data.size())
365 throw_error(ErrorType::FILE_CORRUPTED, "An lz77 header is corrupted in file \"" + file.string() + "\"");
366
367 buffer_.reserve(data.size());
368 buffer_ = std::vector<LZ77_Token>{
369 reinterpret_cast<LZ77_Token const*>(data.data() + len1 + len2),
370 reinterpret_cast<LZ77_Token const*>(data.data() + len1 + len2 + compressed_length_)
371 };
372 }
373

378 * @param output output buffer

383 constexpr size_t len2 = sizeof(decltype(uncompressed_length_));
384
385 output.resize(len1 + len2 + buffer_.size() * sizeof(LZ77_Token));
386
387 std::memcpy(output.data(), &compressed_length_, len1);
388 std::memcpy(output.data() + len1, &uncompressed_length_, len2);
389
390 std::memcpy(output.data() + len1 + len2, buffer_.data(), buffer_.size() * sizeof(LZ77_Token));

449class LZ77Compressor
450{
451public:
452 LZ77Compressor(std::span<Sym const> const data, CompPreset const preset)
453 :data_(data), window(data, preset), pattern_matcher(window, 0, preset)
454 {}

456 /**
457 * @brief Compress the data received in the constructor
458 * @param out_data Where the output will go
459 *
460 * @note Will clear out_data block
461 */
462 template <typename Alloc>
463 void compress(LZ77Block<Alloc>& out_data)
464 {
465 out_data.clear();
466 out_data.reserve(data_.size() / sizeof(LZ77_Token));
467
468 size_t consumed_tokens{};
469 //loop over the stream
470 for (auto i = data_.begin(); i != data_.end();)
471 {
472 auto const token = this->pattern_matcher.find_pattern();
473
474 auto const num_of_rolls = token.length + 1;
475 this->pattern_matcher.roll_hash(num_of_rolls);//roll, updates the hash and position
476 this->window.slide(num_of_rolls);
477 i += num_of_rolls;
478 consumed_tokens += num_of_rolls;
479
480 out_data.push_back(token);
481 }
482 }
483
484private:
485 std::span<Sym const> const data_{};
486 Window window;
487 alg::Rabin pattern_matcher;
488

492template<typename Alloc = std::allocator<LZ77Block<>::value_type>>
493class LZ77Decompressor
494{
495public:
496 explicit LZ77Decompressor(LZ77Block<Alloc> const& data)
497 : data_(data) {}
498
499 /**
500 * @brief Decompress the data received in the constructor
501 * @param out_data Where the output will go
502 *
503 * @note Will clear out_data buffer
504 */
505 template <typename Allocator = std::allocator<Sym>>
506 void decompress(std::vector<Sym, Allocator>& out_data) const
507 {
508 out_data.reserve(data_.size() / sizeof(Sym));
509 for (auto const token : data_)
510 {
511 if (token.offset != 0)
512 {
513 auto size = out_data.size();
514 for (auto _ : std::views::iota(0u, token.length))
515 out_data.push_back(out_data[size - token.offset]);
516 }
517 out_data.emplace_back(token.symbol);
518 }
519 assert(out_data.capacity() == out_data.size());
520 }
521
522private:
523 LZ77Block<Alloc> const& data_;
524};

552 * @brief Run the compression
553 */
554 void compress() noexcept(false)
555 {
556
557 auto const partition_size = std::min(file_size_ / concurrency_, MAX_BLOCK_SIZE);
558 for (size_t completed_size = 0, seq_num{}; completed_size < file_size_; completed_size += partition_size, seq_num++)
559 {
560 //get the true partition size, i.e. the partition can be cut off at the eof
561 auto const true_partition_size = [completed_size, partition_size, this]
562 {
563 if (completed_size + partition_size > file_size_)
564 return file_size_ - completed_size;
565 return partition_size;
566 }();
567
568
569 auto future = thread_pool.add_task([this, completed_size, true_partition_size, seq_num]//'this' shall be strictly used to access const members, or concurrency capable containers
570 {
571 std::ifstream in_file{file_.get_in_file_options().path, std::ios::binary | std::ios::in};
572 in_file.seekg(completed_size);
573
574 std::vector<Sym> data_for_task;
575 data_for_task.resize(true_partition_size);
576
577 in_file.read(reinterpret_cast<char*>(data_for_task.data()), true_partition_size);
578
579 auto out_buffer = std::make_unique<LZ77Block<>>();
580 out_buffer->reserve(true_partition_size);
581
582 LZ77Compressor compressor{{reinterpret_cast<Sym const*>(data_for_task.data()), (data_for_task.size())}, preset_};
583 compressor.compress(*out_buffer);
584 file_list_.insert(std::move(out_buffer), seq_num);
585 });
586
587 std::unique_lock lock{mutex_};
588 results.emplace_back(std::move(future));
589 }
