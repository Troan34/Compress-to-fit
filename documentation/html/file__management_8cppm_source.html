24constexpr std::string_view SIGNATURE = "CTF-1";
25 //SIGNATURE + IDENTIFIER
26export constexpr size_t FILE_HEADER_SIZE = SIGNATURE.size() + sizeof(size_t) + sizeof(uint16_t) + sizeof(uint8_t) + sizeof(uint8_t);
27enum class HEADER_OFFSET
28{
29 SIGNATURE = 0,
30 ID = ::SIGNATURE.size(),
31 COUNT = ID + sizeof(uint16_t),
32 COMP_TYPE = COUNT + sizeof(size_t),
33 COMP_PRESET = COMP_TYPE + sizeof(uint8_t),
34 MAX = COMP_PRESET + sizeof(uint8_t),
35};

37/// TZF FILE HEADER
38/// ___________________________________________________________________________________________________________________
39/// | | | | | |
40/// | SIGNATURE | Identifier (file splitting) (u64) | Count(file splitting)(u16) | CompType(u8) | CompPreset(u8) |
41/// |____________|___________________________________|______________________________|________________|__________________|
42/// ↑ ↑ ↑ ↑ ↑
43/// HEADER_OFFSET::ID HEADER_OFFSET::COUNT /*...*/::COMP_TYPE HEADER_OFFSET::COMP_PRESET HEADER_OFFSET::MAX
44
45//avoid padding in files
46#pragma pack(push, 1)
47struct Header
48{
49 size_t identifier;
50 uint16_t count;
51 CompType comp_type;
52 CompPreset preset;
53};
54#pragma pack(pop)

140 /**
141 * @brief Split an encoded file into a certain number of encoded (with header) files
142 * @param path to be split
143 * @param portions number of files that will be made
144 */
145 static void split_file(fs::path const& path, size_t portions)
146 {
147
148 if (portions == 1) return;
149
150 if ((portions < 1 or portions > N_FILES_LIMIT) or (fs::file_size(path) / portions) < SIZE_FILES_MIN)
151 print_warn(WarningType::PORTIONS_OUT_OF_RANGE, std::to_string(portions));
152
153
154 std::ifstream source_file{ path, std::ios::binary };
155 assert(has_signature(source_file));
156
157 if (!fs::exists(path)) throw_error(ErrorType::PATH_NOT_FOUND, path.string());
158 auto Header = extract_info(path);
159
160 //move to the ID section
161 source_file.seekg(static_cast<size_t>(HEADER_OFFSET::ID));
162 auto portions_size = fs::file_size(path) / portions;
163
164 std::vector<char> buffer(portions_size);
165
166 auto out_path = path.parent_path() / "output";
167 fs::create_directory(out_path);
168
169
170 //Set up a random number such that num < size_t::max
171 std::random_device seed;
172 std::default_random_engine rng(seed());
173 std::uniform_int_distribution<size_t> dist(1, std::numeric_limits<uint16_t>::max());
174 auto id = dist(rng);
175
176 for (uint16_t file_n = 0; file_n < portions; file_n++)
177 {
178 //create a file with same comp_type and preset, but with id as ID. This file ends with _<file_n>
179 auto file = create_file(Header.value().comp_type, Header.value().preset, out_path / (path.stem().string() + '_' + std::to_string(file_n) + FILE_EXTENSION), id, file_n);
180
181 //We are at the last file, may be smaller than portions_size and/or SIZE_FILES_MIN
182 if (portions_size > (fs::file_size(path) - source_file.tellg()))
183 buffer.resize(fs::file_size(path) - source_file.tellg());
184
185 source_file.read(buffer.data(), buffer.size());
186 file.write(buffer.data(), buffer.size());
187 }
188
189 try
190 {
191 switch (errno)
192 {
193 case ENOSPC:
194 case EIO:
195 case ENFILE:
196 case EMFILE:
197 throw_error(ErrorType::DRIVE_ERROR, path.string());
198 default:
199 break;
200 }
201 }
202 catch (std::runtime_error const&)
203 {
204 fs::remove_all(out_path);
205 std::terminate();//terminate as to not call fs::remove(path), also, if these errors do happen it is a good idea to drop everything
206 }
207 source_file.close();
208
209 fs::remove(path);
210 }

218 static void read_file(const fs::path& in_path, std::vector<Sym>& received_data, size_t pos = 0)
219 {

296 /**
297 * @brief Extract information from the header of the tzf file.
298 * @param path Will not be used to construct a stream
299 * @param file The file the information will be extracted from
300 * @return A FileOptions type constructed with the obtained information
301 *
302 * @pre path shall exist and be accessible.
303 */
304 static auto extract_info(std::ifstream& file, fs::path const& path) -> std::optional<Header>
305 {
306
307 //we are dealing with an already compressed file
308 if (has_signature(file))
309 {
310 file.seekg(static_cast<size_t>(HEADER_OFFSET::ID));
311
312 std::byte buffer[sizeof(Header)] = { std::byte{0} };
313 file.read(reinterpret_cast<char*>(buffer), sizeof(Header));
314
315 auto header = std::bit_cast<Header>(buffer);
316
317 if (header.comp_type >= CompType::MAX or header.preset > COMP_MAX)
318 throw_error(ErrorType::FILE_CORRUPTED, path.string());
319
320 try_throw_IO_error(file.rdstate(), path);
321
322 return std::optional{header};
323
324 }
325
326 return std::optional<Header>{};
327 }

329 static auto create_file(CompType const comp_type, CompPreset const comp_preset, fs::path const& path, size_t id, uint16_t count) -> std::ofstream
330 {
331 std::ofstream file(path, std::ios::binary | std::ios::trunc);
332
333 file.seekp(0);
334 file.write(SIGNATURE.data(), SIGNATURE.size());
335 file.write(reinterpret_cast<char const*>(&id), sizeof(decltype(id)));
336 file.write(reinterpret_cast<char const*>(&count), sizeof(decltype(count)));
337 file.seekp(static_cast<int>(HEADER_OFFSET::COMP_PRESET));
338 auto temp = static_cast<uint8_t>(comp_type);
339 file.write(reinterpret_cast<char const*>(&temp), sizeof(decltype(temp)));
340 temp = static_cast<uint8_t>(comp_preset);
341 file.write(reinterpret_cast<char const*>(&temp), sizeof(decltype(temp)));
342
343 return file;
344 }
