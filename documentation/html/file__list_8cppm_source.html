32/**
33 * @brief First sequence in first sequence out, i.e. the data is written in the order given to the 'insert()' member function.
34 * @tparam T Must implement write_to(ofstream&) or be trivially copyable
35 *
36 * @details Implemented as doubly linked list. A separate write thread is spawned.
37 *
38 * @note Because the list will own the elements, consider using a memory pool (e.g. pmr) as to not reallocate memory.
39 * Do NOT expect this stuff to be cache friendly in ANY way.
40 */
41export template<typename T>
42 requires SerializableToDisk<std::ranges::range_value_t<T>> or SerializableToDisk<T>
43class ConcOrderedFileList

45public:

66 /**
67 * @brief Inserts an element to the list, which it will own. THE FIRST ELEMENT MUST HAVE SEQUENCE NUMBER == 0, check warning for additional info.
68 *
69 * @param[in] element Element to be inserted
70 * @param sequence_num The number of this element's sequence
71 *
72 * @pre \p element must not own a nullptr
73 * @post Any reference to \p element is invalidated.
74 *
75 * @warning THE FIRST ELEMENT THAT IS GOING TO BE WRITTEN TO FILE MUST HAVE SEQUENCE NUMBER == 0.
76 * This requirement exists as the first element you add could easily not be the first data chunk (i.e. out of order).
77 * Failure to follow this requirement will cause DEADLOCK.
78 *
79 */
80 void insert(std::unique_ptr<T> element, size_t sequence_num)
81 {
82 auto* new_node = new Node<T>{std::move(element), sequence_num};
83
84 {
85 std::unique_lock lock{mut};//we could opt for a shared mutex here, profile
86
87 size_++;
88
89 if (head_ == nullptr)
90 {
91 head_ = new_node;
92 tail_ = new_node;
93 }
94 else
95 {
96 Node<T>* next_node{head_};
97
98 //Make sure the nullptr check stays first (unless you like bugs, I don't judge)
99 //If sequence_num > next_node->sequence_num_ it means that we found a spot that keeps the ascending order
100 while (next_node and (sequence_num < next_node->sequence_num_)) //walks down the ptrs
101 {
102 next_node = next_node->next;
103 }
104
105 if (!next_node)//We have to become the tail
106 {
107 new_node->prev = tail_;
108 tail_->next = new_node;
109 tail_ = new_node;
110 }
111 else//we either become the head or squeeze in
112 {
113 if (next_node == head_)//become head
114 {
115 new_node->next = head_;
116 head_->prev = new_node;
117 head_ = new_node;
118 }
119 else//squeeze in
120 {
121 //Make the links for the new node to its future neighbors
122 new_node->next = next_node;
123 new_node->prev = next_node->prev;
124
125 //Make the links from the neighbors to the new node
126 next_node->prev->next = new_node;
127 next_node->prev = new_node;
128 }
129 }
130 }
131 called_writer = true;
132 writer_notifier.notify_one();
133 }
134 }

171 /**
172 * @brief Writes to file. Should be called by a thread.
173 * @pre tail_ != nullptr
174 */
175 void write_file(std::stop_token stop)
176 {
177 //NOT thread safe by itself, will not interact with the mutex.
178 //returns true if something has been written otherwise false
179 auto write_func = [this]
180 {
181 if (!tail_)
182 return false;
183
184 if (tail_->sequence_num_ != expected_sequence_num_)//we are missing a node
185 return false;
186
187 write_to_disk(*tail_->element_.get(), file_.get_ref_out_stream());
188
189 auto old_tail = tail_;
190 if (size_ == 1)
191 {
192 tail_ = nullptr;
193 head_ = nullptr;
194 }
195 else
196 {
197 tail_ = tail_->prev;//new tail ptr
198 tail_->next = nullptr;
199 }
200 delete old_tail;//delete old tail ptr
201
202 size_--;
203 expected_sequence_num_++;
204 return true;
205 };
206
207 while (!stop.stop_requested())
208 {
209 std::unique_lock lock{mut};
210 writer_notifier.wait(lock, [this]{return called_writer;});
211
212 if (stop.stop_requested())
213 break;
214 called_writer = false;
215
216 if (size_ > LEAST_NODES_FOR_WRITING)
217 {
218 for (auto _ : std::views::iota(0u, size_))
219 write_func();
220 }
221 }
222
223 std::unique_lock lock{mut};
224 while (write_func());//write the whole thing
225
226 assert(size_ == 0 && "The file list wasn't able to clean up correctly.");
227 }
